not_found.py 
# Oops — route not found
class NotFoundError(Exception):
    status_code = 404
    message = "Resource does not exist"

def handle_request(path):
    routes = [
        "/",
        "/#about",
        "/#stack",
        "/#projects",
        "/#contact"
    ]

    if path not in routes:
        raise NotFoundError(
            f"'{path}' does not exist"
        )

    return "Welcome home"

# → Redirect user back home
handle_request("/")
